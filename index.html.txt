    <script>
        // --- DATA STORE STATE ---
        const APP_STORAGE_KEY = 'chronoflow_planner_app_state';
        
        let state = {
            currentView: 'daily', // 'daily' | 'weekly'
            selectedDailyDate: getFormattedDate(new Date()),
            selectedWeeklyStartDate: getMonday(new Date()),
            dailyData: {}, // Format: { 'YYYY-MM-DD': { priorities: ['', '', ''], todos: [{id, text, completed}], hourly: { '06:00': '' } } }
            weeklyTasks: {} // Format: { 'YYYY-MM-DD': [{ id, text, category, completed }] }
        };

        // --- INITIALIZATION ---
        window.onload = function() {
            loadStateFromLocalStorage();
            initTheme();
            
            // Set date inputs
            document.getElementById('dailyDatePicker').value = state.selectedDailyDate;
            
            renderAll();
        };

        // --- LOCAL STORAGE HANDLING ---
        function loadStateFromLocalStorage() {
            try {
                const saved = localStorage.getItem(APP_STORAGE_KEY);
                if (saved) {
                    const parsed = JSON.parse(saved);
                    state.dailyData = parsed.dailyData || {};
                    state.weeklyTasks = parsed.weeklyTasks || {};
                }
            } catch (e) {
                console.error("Failed to load state from localStorage", e);
            }
        }

        function saveState() {
            try {
                localStorage.setItem(APP_STORAGE_KEY, JSON.stringify({
                    dailyData: state.dailyData,
                    weeklyTasks: state.weeklyTasks
                }));
            } catch (e) {
                console.error("Failed to save state to localStorage", e);
            }
        }

        // --- THEME SYSTEM ---
        function initTheme() {
            const isDark = localStorage.getItem('theme') === 'dark' || 
                (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches);
            if (isDark) {
                document.documentElement.classList.add('dark');
            } else {
                document.documentElement.classList.remove('dark');
            }
        }

        function toggleTheme() {
            if (document.documentElement.classList.contains('dark')) {
                document.documentElement.classList.remove('dark');
                localStorage.setItem('theme', 'light');
            } else {
                document.documentElement.classList.add('dark');
                localStorage.setItem('theme', 'dark');
            }
        }

        // --- VIEW SWITCHING ---
        function switchView(view) {
            state.currentView = view;
            const btnDaily = document.getElementById('btnViewDaily');
            const btnWeekly = document.getElementById('btnViewWeekly');
            const dailySec = document.getElementById('dailyView');
            const weeklySec = document.getElementById('weeklyView');

            if (view === 'daily') {
                dailySec.classList.remove('hidden');
                weeklySec.classList.add('hidden');
                
                btnDaily.className = "px-3 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all-fast flex items-center space-x-2 bg-white dark:bg-zinc-700 text-gray-900 dark:text-white shadow-sm";
                btnWeekly.className = "px-3 py-1.5 rounded-lg text-xs md:text-sm font-medium text-gray-600 dark:text-zinc-400 hover:text-gray-900 dark:hover:text-white transition-all-fast flex items-center space-x-2";
            } else {
                dailySec.classList.add('hidden');
                weeklySec.classList.remove('hidden');
                
                btnWeekly.className = "px-3 py-1.5 rounded-lg text-xs md:text-sm font-medium transition-all-fast flex items-center space-x-2 bg-white dark:bg-zinc-700 text-gray-900 dark:text-white shadow-sm";
                btnDaily.className = "px-3 py-1.5 rounded-lg text-xs md:text-sm font-medium text-gray-600 dark:text-zinc-400 hover:text-gray-900 dark:hover:text-white transition-all-fast flex items-center space-x-2";
            }
            renderAll();
        }

        // --- HELPER DATE FUNCTIONS ---
        function getFormattedDate(dateObj) {
            const year = dateObj.getFullYear();
            const month = String(dateObj.getMonth() + 1).padStart(2, '0');
            const day = String(dateObj.getDate()).padStart(2, '0');
            return `${year}-${month}-${day}`;
        }

        function getMonday(d) {
            const date = new Date(d);
            const day = date.getDay();
            const diff = date.getDate() - day + (day === 0 ? -6 : 1);
            return new Date(date.setDate(diff));
        }

        function formatDateReadable(dateStr) {
            const date = new Date(dateStr + 'T00:00:00');
            return date.toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric', year: 'numeric' });
        }

        function getEnsureDailyObj(dateStr) {
            if (!state.dailyData[dateStr]) {
                state.dailyData[dateStr] = {
                    priorities: ['', '', ''],
                    todos: [],
                    hourly: {}
                };
            }
            return state.dailyData[dateStr];
        }

        // --- RENDER ROUTINER ---
        function renderAll() {
            if (state.currentView === 'daily') {
                renderDailyView();
            } else {
                renderWeeklyView();
            }
        }

        // --- DAILY VIEW RENDERING & CONTROLS ---
        function changeDailyDate(val) {
            if (!val) return;
            state.selectedDailyDate = val;
            renderDailyView();
        }

        function setDailyDateToday() {
            state.selectedDailyDate = getFormattedDate(new Date());
            document.getElementById('dailyDatePicker').value = state.selectedDailyDate;
            renderDailyView();
        }

        function navigateDailyDate(offset) {
            const curr = new Date(state.selectedDailyDate + 'T00:00:00');
            curr.setDate(curr.getDate() + offset);
            state.selectedDailyDate = getFormattedDate(curr);
            document.getElementById('dailyDatePicker').value = state.selectedDailyDate;
            renderDailyView();
        }

        function renderDailyView() {
            const dateStr = state.selectedDailyDate;
            document.getElementById('dailyDateDisplay').innerText = formatDateReadable(dateStr);
            
            const dayData = getEnsureDailyObj(dateStr);

            // 1. Render Top 3 Priorities
            renderPriorities(dayData.priorities);

            // 2. Render To-Dos
            renderTodos(dayData.todos);

            // 3. Render Hourly Grid
            renderHourlyGrid(dayData.hourly);
        }

        function renderPriorities(priorities) {
            const container = document.getElementById('topPrioritiesContainer');
            container.innerHTML = '';

            for (let i = 0; i < 3; i++) {
                const val = priorities[i] || '';
                const itemEl = document.createElement('div');
                itemEl.className = "flex items-center space-x-2";
                itemEl.innerHTML = `
                    <span class="text-xs font-bold w-5 h-5 rounded-full bg-red-50 dark:bg-red-950/80 text-red-600 dark:text-red-400 border border-red-200 dark:border-red-900/60 flex items-center justify-center shrink-0">${i+1}</span>
                    <input type="text" value="${escapeHtml(val)}" placeholder="Priority task #${i+1}" 
                        onchange="updatePriority(${i}, this.value)"
                        class="flex-1 bg-gray-50 dark:bg-zinc-800/80 border border-gray-200 dark:border-zinc-700/80 rounded-xl px-3 py-1.5 text-sm font-medium text-gray-800 dark:text-white focus:outline-none focus:ring-2 focus:ring-red-500/50 transition-all-fast">
                `;
                container.appendChild(itemEl);
            }
        }

        function updatePriority(index, value) {
            const dayData = getEnsureDailyObj(state.selectedDailyDate);
            dayData.priorities[index] = value;
            saveState();
        }

        function renderTodos(todos) {
            const container = document.getElementById('todoList');
            const progressDisplay = document.getElementById('todoProgress');
            const progressBar = document.getElementById('todoProgressBar');
            
            container.innerHTML = '';

            const completedCount = todos.filter(t => t.completed).length;
            const totalCount = todos.length;
            
            progressDisplay.innerText = `${completedCount}/${totalCount} Completed`;
            const pct = totalCount > 0 ? (completedCount / totalCount) * 100 : 0;
            progressBar.style.width = `${pct}%`;

            if (todos.length === 0) {
                container.innerHTML = `<div class="text-center py-6 text-gray-400 dark:text-zinc-600 text-xs italic">No tasks added yet. Add one above!</div>`;
                return;
            }

            todos.forEach((todo, idx) => {
                const item = document.createElement('div');
                item.className = "flex items-center justify-between p-2.5 rounded-xl border border-gray-100 dark:border-zinc-800/80 bg-gray-50/50 dark:bg-zinc-800/40 hover:bg-gray-100/60 dark:hover:bg-zinc-800 transition-all-fast group";
                item.innerHTML = `
                    <div class="flex items-center space-x-3 flex-1 min-w-0 pr-2">
                        <input type="checkbox" ${todo.completed ? 'checked' : ''} onchange="toggleTodo(${todo.id})" class="w-4 h-4 rounded text-blue-600 focus:ring-blue-500 cursor-pointer border-gray-300 dark:border-zinc-700">
                        <span class="text-sm text-gray-800 dark:text-zinc-200 truncate ${todo.completed ? 'line-through text-gray-400 dark:text-zinc-500' : ''}">${escapeHtml(todo.text)}</span>
                    </div>
                    <button onclick="deleteTodo(${todo.id})" class="text-gray-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-all-fast p-1">
                        <i class="fa-solid fa-xmark text-xs"></i>
                    </button>
                `;
                container.appendChild(item);
            });
        }

        function handleAddTodo(e) {
            e.preventDefault();
            const input = document.getElementById('newTodoInput');
            const text = input.value.trim();
            if (!text) return;

            const dayData = getEnsureDailyObj(state.selectedDailyDate);
            dayData.todos.push({
                id: Date.now(),
                text: text,
                completed: false
            });

            input.value = '';
            saveState();
            renderDailyView();
        }

        function toggleTodo(id) {
            const dayData = getEnsureDailyObj(state.selectedDailyDate);
            const todo = dayData.todos.find(t => t.id === id);
            if (todo) {
                todo.completed = !todo.completed;
                saveState();
                renderDailyView();
            }
        }

        function deleteTodo(id) {
            const dayData = getEnsureDailyObj(state.selectedDailyDate);
            dayData.todos = dayData.todos.filter(t => t.id !== id);
            saveState();
            renderDailyView();
        }

        function renderHourlyGrid(hourlyData) {
            const container = document.getElementById('hourlyGrid');
            container.innerHTML = '';

            // Generating hours 06:00 to 23:00
            for (let h = 6; h <= 23; h++) {
                const hourStr = `${String(h).padStart(2, '0')}:00`;
                const val = hourlyData[hourStr] || '';

                const block = document.createElement('div');
                block.className = "flex items-center space-x-3 border-b border-gray-100 dark:border-zinc-800/60 pb-1.5";
                block.innerHTML = `
                    <span class="text-xs font-semibold text-gray-400 dark:text-zinc-500 w-12 text-right shrink-0">${hourStr}</span>
                    <input type="text" value="${escapeHtml(val)}" placeholder="Schedule item..."
                        onchange="updateHourly('${hourStr}', this.value)"
                        class="flex-1 bg-gray-50/50 dark:bg-zinc-800/40 border border-transparent hover:border-gray-200 dark:hover:border-zinc-700 focus:border-blue-500 dark:focus:border-blue-500 focus:bg-white dark:focus:bg-zinc-800 rounded-lg px-3 py-1.5 text-xs font-normal text-gray-800 dark:text-zinc-200 focus:outline-none transition-all-fast">
                `;
                container.appendChild(block);
            }
        }

        function updateHourly(hourStr, value) {
            const dayData = getEnsureDailyObj(state.selectedDailyDate);
            dayData.hourly[hourStr] = value;
            saveState();
        }

        // --- WEEKLY VIEW RENDERING ---
        function setWeeklyDateToday() {
            state.selectedWeeklyStartDate = getMonday(new Date());
            renderWeeklyView();
        }

        function navigateWeeklyDate(offsetDays) {
            const curr = new Date(state.selectedWeeklyStartDate);
            curr.setDate(curr.getDate() + offsetDays);
            state.selectedWeeklyStartDate = getMonday(curr);
            renderWeeklyView();
        }

        function renderWeeklyView() {
            const container = document.getElementById('weeklyGrid');
            container.innerHTML = '';

            const startDate = new Date(state.selectedWeeklyStartDate);
            const endDate = new Date(startDate);
            endDate.setDate(endDate.getDate() + 6);

            document.getElementById('weeklyRangeDisplay').innerText = 
                `${startDate.toLocaleDateString('en-US', { month: 'short', day: 'numeric' })} - ${endDate.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })}`;

            const todayStr = getFormattedDate(new Date());

            for (let i = 0; i < 7; i++) {
                const dayDate = new Date(startDate);
                dayDate.setDate(dayDate.getDate() + i);
                const dayDateStr = getFormattedDate(dayDate);
                const isToday = dayDateStr === todayStr;

                const dayTasks = state.weeklyTasks[dayDateStr] || [];

                const col = document.createElement('div');
                col.className = `bg-white dark:bg-zinc-900 rounded-2xl p-3.5 border ${isToday ? 'border-blue-500 ring-2 ring-blue-500/10' : 'border-gray-200/80 dark:border-zinc-800'} shadow-sm flex flex-col min-h-[420px] transition-all-fast`;
                col.setAttribute('ondragover', 'handleDragOver(event)');
                col.setAttribute('ondragleave', 'handleDragLeave(event)');
                col.setAttribute('ondrop', `handleDrop(event, '${dayDateStr}')`);

                const dayName = dayDate.toLocaleDateString('en-US', { weekday: 'short' });
                const dayNum = dayDate.getDate();

                col.innerHTML = `
                    <div class="flex items-center justify-between pb-3 mb-3 border-b border-gray-100 dark:border-zinc-800">
                        <div class="flex items-center space-x-2">
                            <span class="text-sm font-bold ${isToday ? 'text-blue-600 dark:text-blue-400' : 'text-gray-900 dark:text-white'}">${dayName}</span>
                            <span class="text-xs px-2 py-0.5 rounded-full ${isToday ? 'bg-blue-600 text-white' : 'bg-gray-100 dark:bg-zinc-800 text-gray-500 dark:text-zinc-400'} font-semibold">${dayNum}</span>
                        </div>
                        <button onclick="addWeeklyTaskPrompt('${dayDateStr}')" class="text-xs text-blue-600 hover:text-blue-700 dark:text-blue-400 font-semibold flex items-center space-x-1 p-1 hover:bg-blue-50 dark:hover:bg-blue-950/50 rounded-lg transition-all-fast">
                            <i class="fa-solid fa-plus"></i>
                        </button>
                    </div>

                    <div id="weekly-tasks-${dayDateStr}" class="flex-1 space-y-2 overflow-y-auto max-h-[380px] pr-0.5">
                        ${renderWeeklyTaskItems(dayTasks, dayDateStr)}
                    </div>
                `;

                container.appendChild(col);
            }
        }

        function renderWeeklyTaskItems(tasks, dayDateStr) {
            if (tasks.length === 0) {
                return `<div class="text-center py-8 text-gray-300 dark:text-zinc-700 text-xs italic">No tasks</div>`;
            }

            return tasks.map(task => `
                <div id="task-${task.id}" draggable="true" ondragstart="handleDragStart(event, '${task.id}', '${dayDateStr}')" 
                    class="p-2.5 rounded-xl bg-gray-50 dark:bg-zinc-800/80 border border-gray-200/80 dark:border-zinc-700/60 shadow-sm cursor-grab active:cursor-grabbing hover:border-gray-300 dark:hover:border-zinc-600 transition-all-fast group">
                    <div class="flex items-start justify-between gap-1.5">
                        <div class="flex items-start space-x-2 flex-1 min-w-0">
                            <input type="checkbox" ${task.completed ? 'checked' : ''} onchange="toggleWeeklyTask('${dayDateStr}', ${task.id})" class="mt-0.5 w-3.5 h-3.5 rounded text-indigo-600 focus:ring-indigo-500 cursor-pointer border-gray-300 dark:border-zinc-700">
                            <span class="text-xs text-gray-800 dark:text-zinc-200 break-words leading-tight ${task.completed ? 'line-through text-gray-400 dark:text-zinc-500' : ''}">${escapeHtml(task.text)}</span>
                        </div>
                        <button onclick="deleteWeeklyTask('${dayDateStr}', ${task.id})" class="text-gray-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-all-fast p-0.5">
                            <i class="fa-solid fa-xmark text-xs"></i>
                        </button>
                    </div>
                </div>
            `).join('');
        }

        function addWeeklyTaskPrompt(dayDateStr) {
            const text = prompt("Enter weekly task:");
            if (!text || !text.trim()) return;

            if (!state.weeklyTasks[dayDateStr]) {
                state.weeklyTasks[dayDateStr] = [];
            }

            state.weeklyTasks[dayDateStr].push({
                id: Date.now(),
                text: text.trim(),
                completed: false
            });

            saveState();
            renderWeeklyView();
        }

        function toggleWeeklyTask(dayDateStr, taskId) {
            if (state.weeklyTasks[dayDateStr]) {
                const task = state.weeklyTasks[dayDateStr].find(t => t.id === taskId);
                if (task) {
                    task.completed = !task.completed;
                    saveState();
                    renderWeeklyView();
                }
            }
        }

        function deleteWeeklyTask(dayDateStr, taskId) {
            if (state.weeklyTasks[dayDateStr]) {
                state.weeklyTasks[dayDateStr] = state.weeklyTasks[dayDateStr].filter(t => t.id !== taskId);
                saveState();
                renderWeeklyView();
            }
        }

        // --- DRAG AND DROP HANDLERS ---
        let draggedTaskData = null;

        function handleDragStart(e, taskId, sourceDayStr) {
            draggedTaskData = { taskId: parseInt(taskId), sourceDayStr };
            e.dataTransfer.effectAllowed = 'move';
            e.target.classList.add('dragging');
        }

        function handleDragOver(e) {
            e.preventDefault();
            e.dataTransfer.dropEffect = 'move';
            e.currentTarget.classList.add('drag-over');
        }

        function handleDragLeave(e) {
            e.currentTarget.classList.remove('drag-over');
        }

        function handleDrop(e, targetDayStr) {
            e.preventDefault();
            e.currentTarget.classList.remove('drag-over');

            if (!draggedTaskData) return;

            const { taskId, sourceDayStr } = draggedTaskData;

            if (sourceDayStr === targetDayStr) return;

            // Find and move task
            const sourceList = state.weeklyTasks[sourceDayStr] || [];
            const taskIndex = sourceList.findIndex(t => t.id === taskId);

            if (taskIndex > -1) {
                const [task] = sourceList.splice(taskIndex, 1);

                if (!state.weeklyTasks[targetDayStr]) {
                    state.weeklyTasks[targetDayStr] = [];
                }

                state.weeklyTasks[targetDayStr].push(task);
                saveState();
                renderWeeklyView();
            }

            draggedTaskData = null;
        }

        // --- RESET DATA SYSTEM ---
        function confirmReset() {
            document.getElementById('resetModal').classList.remove('hidden');
        }

        function closeResetModal() {
            document.getElementById('resetModal').classList.add('hidden');
        }

        function executeReset() {
            localStorage.removeItem(APP_STORAGE_KEY);
            state.dailyData = {};
            state.weeklyTasks = {};
            closeResetModal();
            renderAll();
        }

        // --- UTILS ---
        function escapeHtml(str) {
            if (!str) return '';
            return str
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
